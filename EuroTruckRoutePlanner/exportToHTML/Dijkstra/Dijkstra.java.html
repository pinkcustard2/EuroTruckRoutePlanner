15       private NodeConnections<Integer> findNode(int key)
16       {
17           // finds a node with a given key
18           NodeConnections<Integer> nodeConnectionsChecked = null;
19   
20           for(int i = 0; i < graph.getMaxSize(); i++) // loops through graph
21           {
22               if(graph.item(i) != null && key == i) // checks if node being checked key is equal to the key argument
23               {
24                   nodeConnectionsChecked = graph.item(i); // node found
25                   break;
26               }
27           }
28   
29           return nodeConnectionsChecked;
30       }

37       public int pathLength(int startNode, int endNode)
38       {
39           // returns shortest path length from start node to end node
40           priorityQueue.add(graph.index(startNode), 0); // add start node to priority queue
41   
42           for(int i = 0; i < graph.getMaxSize(); i++) // loop through graph
43           {
44               if(graph.item(i) != null && i != graph.index(startNode)) // add the rest of the nodes to the priority queue
45               {
46                   priorityQueue.add(i, 1000000);
47               }
48           }
49   
50           while(!priorityQueue.isEmpty())
51           {
52               int currentNode = priorityQueue.pop(); // get the current node with the lowest priority
53               NodeConnections<Integer> nodeConnectionsChecked = this.findNode(currentNode); // get the node connections for the node with the lowest priority
54   
55               for(int i = 0; i < nodeConnectionsChecked.getMaxSize(); i++) // loop through all arcs for current node
56               {
57                   if(nodeConnectionsChecked.getNodeConnections()[i] != null) // arc found
58                   {
59                       int location = nodeConnectionsChecked.getNodeConnections()[i].getValue(); // get the node the arc is travelling to
60                       location = priorityQueue.index(location); // find where in priority queue node the arc is travelling to is
61   
62                       if(location != -1) // node arc is travelling to has not yet been fully explored so is still in the priority queue
63                       {
64                           int arcEndNode = priorityQueue.pop(location); // get the key of the node the arc is travelling to
65   
66                           if(graph.item(arcEndNode).getTempDistance() == 0 || graph.item(arcEndNode).getTempDistance() > graph.item(currentNode).getTempDistance() + nodeConnectionsChecked.getNodeConnections()[i].getKey())
67                           {
68                               // either node has not yet been visited or a shorter path to the node has been found than the previous shortest path
69                               // so the new shorter distance should be set to the temporary distance to that node from the start node
70                               graph.item(arcEndNode).setTempDistance(graph.item(currentNode).getTempDistance() + nodeConnectionsChecked.getNodeConnections()[i].getKey());
71                           }
72                           priorityQueue.add(arcEndNode, graph.item(arcEndNode).getTempDistance()); // re-add the new node with the new distance as the priority
73                       }
74                   }
75               }
76           }
77   
78           return graph.item(endNode).getTempDistance(); // return the final distance of the end node
79       }

81       public ArrayList<Integer> path(int startNode, int endNode)
82       {
83           // returns nodes travelled for the shortest path from the start node to the end node
